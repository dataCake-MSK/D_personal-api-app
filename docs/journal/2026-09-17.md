### 핵심 명령
```bash
# 스캐폴딩 (create-expo-app은 빈 폴더 필요 → 임시 폴더에서 생성 후 복사)
npx create-expo-app@latest expo-app --yes          # SDK 57, TS, expo-router(src/app)
cd expo-app && echo n | node scripts/reset-project.js   # 예제 삭제, 빈 index/_layout 생성
cp -r app.json package.json package-lock.json tsconfig.json assets src <저장소>/
cd <저장소> && npm ci
npx expo install --check                           # SDK 호환 버전 확인
npx tsc --noEmit

# 폰 미리보기
npm i -D @expo/ngrok@^4.1.0
npx expo login                                     # PC 별도 터미널, Expo Go와 같은 계정
npx expo start --tunnel --go                       # Expo Go 홈 → 개발 서버 목록에서 열기
curl -s http://127.0.0.1:4040/api/tunnels          # 백그라운드 실행 시 공개 주소 확인

# PR 머지·브랜치 정리
gh pr merge 12 --merge --delete-branch
git push origin --delete <머지된-브랜치>
git fetch --prune

# 다른 작업 중 변경분 잠시 보관/복원
git stash push -u -m "설명" && git stash pop

# 폴더 이름 변경(이력 유지)
git mv docs/issues docs/reports
```
