5. CI 준비: gh 토큰에 `workflow` 권한 추가

### 핵심 명령
```bash
git init -b main

# GitHub CLI 설치·로그인·저장소 생성
winget install --id GitHub.cli -e
gh auth login --web --git-protocol https
gh auth setup-git
gh repo create dataCake-MSK/D_personal-api-app --public --source . --remote origin --push
gh auth refresh -h github.com -s workflow      # .github/workflows push 권한

# 비밀 값 유출 방지
R=dataCake-MSK/D_personal-api-app
gh api -X PATCH repos/$R \
  -f 'security_and_analysis[secret_scanning][status]=enabled' \
  -f 'security_and_analysis[secret_scanning_push_protection][status]=enabled'

# 라벨·마일스톤
gh label create "type:feature" --color 1f883d --description "기능 구현" -R $R --force
gh api repos/$R/milestones -f title="M0 기반" -f description="앱 스캐폴딩, 품질 도구, CI"

# SRS 항목 → 이슈
gh issue create -R $R --title "[SRS-001] Expo 앱 스캐폴딩" --body-file SRS-001.md \
  --label type:chore --label area:infra --milestone "M0 기반"
```
